import uvicorn
from fastapi import FastAPI, WebSocket, WebSocketDisconnect, Request, Response
from fastapi.middleware.cors import CORSMiddleware
import asyncio, json, subprocess, time, re, threading

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"])

generator_socket = None
pending_requests = {}
req_id = 0

@app.get("/")
async def root():
    return {"status": "Perchance Generator Proxy Bridge Active"}

@app.websocket("/")
async def websocket_endpoint(websocket: WebSocket):
    global generator_socket
    await websocket.accept()
    generator_socket = websocket
    try:
        while True:
            data = await websocket.receive_text()
            msg = json.loads(data)
            msg_id = msg.get("id")
            if msg_id in pending_requests:
                pending_requests[msg_id].set_result(msg.get("body"))
    except WebSocketDisconnect:
        if generator_socket == websocket:
            generator_socket = None

@app.post("/v1/chat/completions")
async def chat_completions(request: Request):
    global generator_socket, req_id
    if not generator_socket:
        return Response(content=json.dumps({"error": "Generator offline"}), status_code=503, media_type="application/json")
    body = await request.json()
    req_id += 1
    current_id = req_id
    loop = asyncio.get_event_loop()
    future = loop.create_future()
    pending_requests[current_id] = future
    await generator_socket.send_text(json.dumps({"id": current_id, "body": body}))
    try:
        return await asyncio.wait_for(future, timeout=30.0)
    except asyncio.TimeoutError:
        return Response(content=json.dumps({"error": "Timeout"}), status_code=504, media_type="application/json")
    finally:
        pending_requests.pop(current_id, None)

const http = require('http');
const { WebSocketServer } = require('ws');
const PORT = process.env.PORT || 3000;
let genSocket = null;
const pending = new Map();
let reqId = 0;

const server = http.createServer(async (req, res) => {
  res.setHeader('Access-Control-Allow-Origin', '*');
  res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization');
  if (req.method === 'OPTIONS') { res.writeHead(204); res.end(); return; }
  if (req.method === 'POST' && req.url.endsWith('/v1/chat/completions')) {
    if (!genSocket) { res.writeHead(503); res.end(JSON.stringify({error:'Generator offline'})); return; }
    let body = ''; req.on('data', c => body += c);
    req.on('end', async () => {
      const id = ++reqId;
      genSocket.send(JSON.stringify({ id, body: JSON.parse(body) }));
      const result = await new Promise(resolve => pending.set(id, { resolve }));
      res.writeHead(200, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify(result));
    });
    return;
  }
  res.writeHead(200); res.end('Perchance Generator Proxy Active.');
});

let genSocket = null;
const pending = new Map();
let reqId = 0;
const cors = {"Access-Control-Allow-Origin":"*","Access-Control-Allow-Methods":"GET,POST,OPTIONS","Access-Control-Allow-Headers":"Content-Type,Authorization"};

export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    if (request.method === "OPTIONS") return new Response(null, { headers: cors });
    const upgrade = request.headers.get("Upgrade");
    if (upgrade?.toLowerCase() === "websocket") {
      const pair = new WebSocketPair();
      const [client, server] = Object.values(pair);
      server.accept(); genSocket = server;
      server.addEventListener("message", e => {
        const { id, body } = JSON.parse(e.data);
        if (pending.has(id)) { pending.get(id).resolve(body); pending.delete(id); }
      });
      server.addEventListener("close", () => { if (genSocket === server) genSocket = null; });
      return new Response(null, { status: 101, webSocket: client });
    }
    if (request.method === "POST" && url.pathname.endsWith("/v1/chat/completions")) {
      if (!genSocket) return new Response(JSON.stringify({error:"Generator offline"}),{status:503,headers:cors});
      const id = ++reqId;
      const body = await request.json();
      genSocket.send(JSON.stringify({ id, body }));
      const result = await new Promise(resolve => pending.set(id, { resolve }));
      return new Response(JSON.stringify(result), { headers: { "Content-Type": "application/json", ...cors } });
    }
    return new Response("Perchance Generator Proxy Active.", { headers: cors });
  }
};
